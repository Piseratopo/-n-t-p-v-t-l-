\exercise Cho hai tập hợp khác rỗng $B$ và $K$, và ánh xạ $a: B \to K$. Chứng minh rằng
\begin{enumerate}
   \item $a$ đơn ánh khi và chỉ khi tồn tại một toàn ánh $b$ từ $B$ đến $K$ sao cho $b \circ a = \mathds{1}_B$,
   \item $a$ toàn ánh khi và chỉ khi tồn tại một đơn ánh $c$ từ $B$ đến $K$ sao cho $a \circ c = \mathds{1}_A$.
\end{enumerate}

\exercise Cho $E$ là một tập hợp, $n \in \mathbb{Z}^+$, $A_0, \dots, A_n$ là những bộ phận của $E$ sao cho:
\[
    \varnothing = A_0 \subsetneq A_1 \subsetneq A_2 \dots \subsetneq A_n = E.
\]
Kí hiệu $B_1 = A_1 \setminus A_0, \dots, B_n = A_n \setminus A_{n-1}$. Chứng minh rằng $\{B_1; \dots; B_n\}$ là một phân hoạch của $E$.

\exercise Cho các tập $A$, $B$, $C$ thỏa mãn $A \in B$ và $B \in C$. Chứng minh rằng $C \notin A$ và $C \nsubseteq A$. Mở rộng với chuỗi hữu hạn các tập hợp.

\exercise Cho $E$ là một tập hợp, $n$ là số nguyên dương ($n \ge 2$) và $A_1, A_2, \dots, A_n$ là các tập con của $E$. Với mỗi ánh xạ dưới đây, hãy tìm điều kiện cần và đủ của họ các tập hợp $(A_i)_{i=1}^n$ để ánh xạ đó là đơn ánh, và điều kiện cần và đủ để ánh xạ đó là toàn ánh:
\begin{enumerate}
   \item Ánh xạ $f_1 : \mathfrak{P}(E) \to \prod_{i=1}^n \mathfrak{P}(A_i)$ xác định bởi
   \begin{equation*}
      \forall x \in \mathfrak{P}(E), \quad f_1(x) = (x \cap A_1,\, x \cap A_2,\, \dots,\, x \cap A_n);
   \end{equation*}

   \item Ánh xạ $f_2 : \mathfrak{P}(E) \to \prod_{i=1}^n \mathfrak{P}(E \setminus A_i)$ xác định bởi
   \begin{equation*}
      \forall x \in \mathfrak{P}(E), \quad f_2(x) = (x \setminus A_1,\, x \setminus A_2,\, \dots,\, x \setminus A_n);
   \end{equation*}

   \item Ánh xạ $f_3 : \mathfrak{P}(E) \to \prod_{i=1}^n \mathfrak{P}(A_i)$ xác định bởi
   \begin{equation*}
      \forall x \in \mathfrak{P}(E), \quad f_3(x) = (A_1 \setminus x,\, A_2 \setminus x,\, \dots,\, A_n \setminus x).
   \end{equation*}
\end{enumerate}

\exercise Cho $K$, $L$, $X$ là các tập hợp. Đặt $I$ là một ánh xạ từ $L$ vào $\mathfrak{P}(K)$ và $f$ là một ánh xạ từ $K$ vào $\mathfrak{P}(X)$. Chứng minh rằng
\begin{multicols}{2}
   \begin{enumerate}
      \item $\displaystyle \bigcup_{l \in L} \left(\bigcap_{i \in I(l)} f(i)\right) = \bigcap_{j \in \prod_{l \in L} I(l)} \left(\bigcup_{l \in L} f(j(l))\right)$;
      \item $\displaystyle \bigcap_{l \in L} \left(\bigcup_{i \in I(l)} f(i)\right) = \bigcup_{j \in \prod_{l \in L} I(l)} \left(\bigcap_{l \in L} f(j(l))\right)$.
   \end{enumerate}
\end{multicols}
   
\solution

\resetSubexercise Gọi $e$ là một phần tử thuộc $\bigcup_{l \in L} \left(\bigcap_{i \in I(l)} f(i)\right)$. Tồn tại $\lambda \in L$ sao cho $e \in \bigcap_{i \in I(\lambda)}f(i)$, dẫn đến \begin{equation}
   \forall i \in I(\lambda), e \in f(i). 
   \label{eq:li_thuyet_tap_hop:tien_de:ho:bt_giao_hop_tich_1}
\end{equation}

Xét $j$ tuỳ ý trong $\prod_{l \in L}I(l)$. Từ định nghĩa, cần phải có $\forall l \in L, j(l) \in I(l)$, kéo theo $j(\lambda) \in I(\lambda)$. Sử dụng \eqref{eq:li_thuyet_tap_hop:tien_de:ho:bt_giao_hop_tich_1}, $e \in f(j(\lambda))$. Qua đó, chúng ta có được $e \in \bigcup_{l \in L} f(j(l))$. Khẳng định này đúng với mọi $j$ ($j$ tùy ý), nên $e \in \bigcap_{j \in \prod_{l \in L}I(l)} \left(\bigcup_{l \in L}f(j(l))\right)$. Kết luận được, $$
   \bigcup_{l \in L}\left(\bigcap_{i \in I(l)}f(i)\right) \subseteq \bigcap_{j \in \prod_{l \in L}I(l)} \left(\bigcup_{l \in L}f(j(l))\right). 
$$
